            || alias_record->effect_observed != 1 || alias_record->restore_attempted
            || alias_record->restore_returned || alias_record->restore_effect_observed) { break; }
    alias_inner = PyMethod_GET_FUNCTION(alias_mutation->bound_owned);

    alias_node = alias_state->controls.record_head;
    while (alias_node != NULL) {

                || Py_TYPE((PyObject *)alias_node) != (PyTypeObject *)alias_state->arena.install.record_type
                || alias_node->linked != 1 || alias_node->registry_owner_state != alias_state
                || alias_node->origin_state != alias_state || alias_node->prev != alias_prior
                || alias_node->module_lifetime != source->module
                || alias_node->owner_proxy != (PyObject *)source->owner
                || alias_node->retired_error_class != alias_state->arena.known.roots[97]
                || alias_node->original_loader == NULL || alias_node->loader_dict == NULL
                || !PyDict_Check(alias_node->loader_dict)
                || alias_node->attribute_key == NULL
                || !exact_ascii_name(alias_node->attribute_key, "exec_module", 11)
                || alias_node->phase != RENO_LOADER_RECORD_ACTIVE || alias_node->restore_cancelled

             * Every queried or previous-only node passes the complete gates.

                        || alias_same_current || alias_same_candidate))
                    || alias_node->store_attempted != 1 || alias_node->store_returned != 1
                    || alias_node->effect_observed != 1 || alias_node->restore_attempted
                    || alias_node->restore_returned || alias_node->restore_effect_observed
                    || alias_node->installed_bound == NULL
                    || !Py_IS_TYPE(alias_node->installed_bound, &PyMethod_Type)
                    || !reno_loader_binding_source_fresh_install_raw(source, alias_node->original_loader)) {
                alias_scan_ok = 0;
                break;

            if (!reno_loader_created_dispatch_matches_raw(alias_node, alias_node->original_loader,

                alias_scan_ok = 0;
                break;
